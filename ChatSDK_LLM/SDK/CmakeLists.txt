cmake_minimum_required(VERSION 3.10)

#设置sdk名称
set(SDK_NAME "ai_chat_sdk")

#设置C++标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

#设置构建类型
set(CMAKE_BUILD_TYPE Release)

#收集源文件
file(GLOB_RECURSE SDK_SOURCES "C_source/*.cpp")
file(GLOB_RECURSE SDK_HEADERS "include/*.h")

#生成静态库
add_library(${SDK_NAME} STATIC ${SDK_SOURCES})

#设置输出目录
set_target_properties(${SDK_NAME} PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/build"
)


#定义httplib中的CPPHTTPLIB_OPENSSL_SUPPORT
target_compile_definitions(${SDK_NAME} PUBLIC CPPHTTPLIB_OPENSSL_SUPPORT)

target_include_directories(${SDK_NAME} PUBLIC 
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

#设置静态库的目录
link_directories(/usr/local/lib)

find_package(OpenSSL REQUIRED)

#添加CPPHTTPLIB_OPENSSL_SUPPORT的定义
target_compile_definitions(${SDK_NAME} PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)


#链接库
target_link_libraries(${SDK_NAME} jsoncpp fmt spdlog sqlit3 
OpenSSL::Crypto OpenSSL::SSL)

#安装规则：拷贝静态库到系统目录下 /usr/local/lib
install(TARGETS ${SDK_NAME} ARCHIVE DESTINATION lib)

#安装规则：拷贝头文件到系统目录下 /usr/local/include
install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/include/ 
DESTINATION include/${SDK_NAME}
Files_MATCHING PATTERN "*.h"
)
